// Use DBML to define your database structure
// Docs: https://dbml.dbdiagram.io/docs

Table admins {
  id UUID [PRIMARY KEY]
  name VARCHAR(100) [NOT NULL]
  email VARCHAR(255) [NOT NULL, UNIQUE]
  password VARCHAR(255) [NOT NULL]
  role VARCHAR(64) [NOT NULL]
  created_at TIMESTAMP [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMP
  deleted_at TIMESTAMP
}

Table donors {
  id UUID [PRIMARY KEY]
  name VARCHAR(64) [NOT NULL]
  email VARCHAR(128) [NOT NULL, UNIQUE]
  phone VARCHAR(20) [NOT NULL, UNIQUE]
  password VARCHAR(255) [NOT NULL]
  created_at TIMESTAMP [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMP
  deleted_at TIMESTAMP
}

TABLE foundations {
  id UUID [PRIMARY KEY]
  name VARCHAR(255) [NOT NULL]
  description TEXT
  address VARCHAR(255)
  slug VARCHAR(255) [NOT NULL, UNIQUE]
  logo_url VARCHAR(500)
  admin_id UUID [NOT NULL]
  created_at TIMESTAMP [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMP
  deleted_at TIMESTAMP
}

Table campaigns {
  id UUID [PRIMARY KEY]
  foundation_id UUID [NOT NULL]
  title VARCHAR(255) [NOT NULL]
  slug VARCHAR(255) [NOT NULL, UNIQUE]
  description TEXT
  target_amount NUMERIC(15, 2) [NOT NULL]
  current_amount NUMERIC(15, 2) [NOT NULL]
  status VARCHAR(50) [NOT NULL, note: "draft, active, completed, cancelled"]
  start_date TIMESTAMP [NOT NULL]
  end_date TIMESTAMP [NOT NULL]
  created_by UUID [NOT NULL]
  created_at TIMESTAMP [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMP
  deleted_at TIMESTAMP
}

TABLE donations {
    id UUID [PRIMARY KEY]
    donor_id UUID [NOT NULL]
    campaign_id UUID [NOT NULL]
    invoice_number VARCHAR(100) [NOT NULL, UNIQUE]
    amount NUMERIC(15, 2) [NOT NULL]
    status VARCHAR(50) [NOT NULL, note: "pending, success, failed, expired"]
    message TEXT
    created_at TIMESTAMP [default: `CURRENT_TIMESTAMP`]
    updated_at TIMESTAMP
    deleted_at TIMESTAMP
}

TABLE payment_logs {
    id UUID [PRIMARY KEY]
    donation_id UUID [NOT NULL]
    gateway_name VARCHAR(50) [NOT NULL] // Misal: Midtrans, Xendit, Stripe
    payment_method VARCHAR(50) // Misal: qris, bca_va, gopay
    paid_at TIMESTAMP
    created_at TIMESTAMP [default: `CURRENT_TIMESTAMP`]
    updated_at TIMESTAMP
    deleted_at TIMESTAMP
}

// Foreign Key Relations
ref: campaigns.created_by > admins.id
ref: donations.donor_id > donors.id
ref: donations.campaign_id > campaigns.id
ref: campaigns.foundation_id > foundations.id
ref: foundations.admin_id > admins.id
ref: payment_logs.donation_id > donations.id
